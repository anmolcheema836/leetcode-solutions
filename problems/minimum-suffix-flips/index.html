1529. Minimum Suffix Flips

class Solution {
 public:
  int minFlips(string target) {
    int ans = 0;
    int state = 0;

    for (const char c : target)
      if (c - '0' != state) {
        state = c - '0';
        ++ans;
      }

    return ans;
  }
};
class Solution {
  public int minFlips(String target) {
    int ans = 0;
    int state = 0;

    for (final char c : target.toCharArray())
      if (c - '0' != state) {
        state = c - '0';
        ++ans;
      }

    return ans;
  }
}
