class DataStream {
 public:
  DataStream(int value, int k) : value(value), k(k) {}

  bool consec(int num) {
    if (q.size() == k) {
      if (q.front() == value)
        --count;
      q.pop();
    }
    if (num == value)
      ++count;
    q.push(num);
    return count == k;
  }

 private:
  const int value;
  const int k;
  queue<int> q;
  int count = 0;
};
class DataStream {
  public DataStream(int value, int k) {
    this.value = value;
    this.k = k;
  }

  public boolean consec(int num) {
    if (q.size() == k && q.poll() == value)
      --count;
    if (num == value)
      ++count;
    q.offer(num);
    return count == k;
  }

  private int value;
  private int k;
  private Queue<Integer> q = new ArrayDeque<>();
  private int count = 0;
}

    self.k = k

    self.count = 0

  def consec(self, num: int) -> bool:
    if len(self.q) == self.k and self.q.popleft() == self.value:
      self.count -= 1
    if num == self.value:
      self.count += 1
