class Solution {
 public:
  long long minOperationsToMakeMedianK(vector<int>& nums, int k) {
    const int n = nums.size();
    long ans = 0;

    ranges::sort(nums);

    for (int i = 0; i <= n / 2; ++i)
      ans += max(0, nums[i] - k);

    for (int i = n / 2; i < n; ++i)
      ans += max(0, k - nums[i]);

    return ans;
  }
};
class Solution {
  public long minOperationsToMakeMedianK(int[] nums, int k) {
    final int n = nums.length;
    long ans = 0;

    Arrays.sort(nums);

    for (int i = 0; i <= n / 2; ++i)
      ans += Math.max(0, nums[i] - k);

    for (int i = n / 2; i < n; ++i)
      ans += Math.max(0, k - nums[i]);

    return ans;
  }
}
class Solution:
  def minOperationsToMakeMedianK(self, nums: list[int], k: int) -> int:
    n = len(nums)
    ans = 0

    nums.sort()

    for i in range(n // 2 + 1):
      ans += max(0, nums[i] - k)

    for i in range(n // 2, n):
      ans += max(0, k - nums[i])

    return ans
