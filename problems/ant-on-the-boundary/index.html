    return ranges::count(nums, 0);

class Solution {
  public int returnToBoundaryCount(int[] nums) {
    int ans = 0;
    int prefix = 0;

    for (final int num : nums) {
      prefix += num;
      if (prefix == 0)
        ++ans;
    }

    return ans;
  }
}
class Solution:
  def returnToBoundaryCount(self, nums: list[int]) -> int:
    return sum(prefix == 0 for prefix in itertools.accumulate(nums))
