class Solution {
 public:
  string baseNeg2(int n) {
    string ans;

    while (n != 0) {
      ans += to_string(n % 2);
      n = -(n >> 1);
    }

    return ans.empty() ? "0" : string{ans.rbegin(), ans.rend()};
  }
};
class Solution {
  public String baseNeg2(int n) {
    StringBuilder sb = new StringBuilder();

    while (n != 0) {
      sb.append(n % 2);
      n = -(n >> 1);
    }

    return sb.length() > 0 ? sb.reverse().toString() : "0";
  }
}
class Solution:
  def baseNeg2(self, n: int) -> str:
    ans = []
