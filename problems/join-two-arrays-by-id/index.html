2722. Join Two Arrays by ID

Intuition & Approach

To solve Join Two Arrays by ID efficiently, we can use standard algorithms and data structures. Explaining the logic, time complexity, and space complexity here will help you understand the solution better.

Code Solution

TypeScript
type JSONValue =
  | {}
  | null
  | boolean
  | number
  | string
  | JSONValue[]
  | { [key: string]: JSONValue };

type Item = { [key: string]: JSONValue } & {
  id: number;
};

function join(arr1: Item[], arr2: Item[]): Item[] {
  const idToObj: { [key: number]: Item } = {};
  for (const item of arr1) {
    idToObj[item.id] = item;
  }
  for (const item of arr2) {
    if (idToObj[item.id] === undefined) {
      idToObj[item.id] = item;
    } else {
      for (const key of Object.keys(item)) {
        idToObj[item.id][key] = item[key];
      }
    }
  }
  const ans = Object.values(idToObj);
  ans.sort((a, b) => a.id - b.id);
  return ans;
}
