class Solution {
 public:
  long long maximumImportance(int n, vector<vector<int>>& roads) {
    long ans = 0;
    vector<long> count(n);

    for (const vector<int>& road : roads) {
      const int u = road[0];
      const int v = road[1];
      ++count[u];
      ++count[v];
    }

    ranges::sort(count);

    for (int i = 0; i < n; ++i)
      ans += (i + 1) * count[i];

    return ans;
  }
};
class Solution {
  public long maximumImportance(int n, int[][] roads) {
    long ans = 0;
    long[] count = new long[n];

    for (int[] road : roads) {
      final int u = road[0];
      final int v = road[1];
      ++count[u];
      ++count[v];
    }

    Arrays.sort(count);

    for (int i = 0; i < n; ++i)
      ans += (i + 1) * count[i];

    return ans;
  }
}
class Solution:
  def maximumImportance(self, n: int, roads: list[list[int]]) -> int:
    count = [0] * n

    for u, v in roads:
      count[u] += 1
      count[v] += 1

    count.sort()
    return sum((i + 1) * c for i, c in enumerate(count))
