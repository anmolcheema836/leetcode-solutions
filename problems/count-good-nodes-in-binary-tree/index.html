1448. Count Good Nodes in Binary Tree

To solve Count Good Nodes in Binary Tree efficiently, we can use standard algorithms and data structures. Explaining the logic, time complexity, and space complexity here will help you understand the solution better.

class Solution {
 public:
  int goodNodes(TreeNode* root, int mx = INT_MIN) {
    if (root == nullptr)
      return 0;
    const int newMax = max(mx, root->val);
    return (root->val >= mx) +              //
           goodNodes(root->left, newMax) +  //
           goodNodes(root->right, newMax);
  }
};
class Solution {
  public int goodNodes(TreeNode root) {
    return goodNodes(root, Integer.MIN_VALUE);
  }

  private int goodNodes(TreeNode root, int mx) {
    if (root == null)
      return 0;

    final int newMax = Math.max(mx, root.val);
    return (root.val >= mx ? 1 : 0) + goodNodes(root.left, newMax) + goodNodes(root.right, newMax);
  }
}
